const express = require('express');
const bcrypt = require('bcryptjs');
const { PrismaClient } = require('@prisma/client');
const { v4: uuid } = require('uuid');
const router = express.Router();
const prisma = new PrismaClient();
const { auth, requireAdmin } = require('../middleware/auth');
const { sendVerificationCode } = require('../utils/mailer');
const { logActivity } = require('../utils/security');
const { validatePassword } = require('../middleware/passwordPolicy');

const pendingCodes = new Map();

// ADMIN: Generate employee registration link
router.post('/employee/generate', auth, requireAdmin, async (req, res) => {
  try {
    const token = uuid();
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);

    await prisma.employeeInvite.create({
      data: { token, createdBy: req.user.id, expiresAt },
    });

    await logActivity({
      userId: req.user.id,
      action: 'EMPLOYEE_INVITE_GENERATE',
      interface: 'web',
      details: token,
      ip: req.ip,
    });

    res.json({
      token,
      expiresAt,
      url: `${process.env.FRONTEND_URL}/employee/register/${token}`,
    });
  } catch (err) {
    console.error('Generate invite error:', err);
    res.status(500).json({ error: err.message });
  }
});

// Verify invite link
router.get('/employee/verify/:token', async (req, res) => {
  const invite = await prisma.employeeInvite.findUnique({
    where: { token: req.params.token },
  });
  if (!invite || !invite.active || invite.expiresAt < new Date()) {
    return res.status(410).json({ error: 'Registration link expired or already used' });
  }
  res.json({ valid: true, expiresAt: invite.expiresAt });
});

// Send 6-digit code to email
router.post('/employee/send-code', async (req, res) => {
  try {
    const { email, token } = req.body;
    if (!email || !token) return res.status(400).json({ error: 'Email and token required' });

    const invite = await prisma.employeeInvite.findUnique({ where: { token } });
    if (!invite || !invite.active || invite.expiresAt < new Date()) {
      return res.status(410).json({ error: 'Registration link expired' });
    }

    const existing = await prisma.user.findUnique({ where: { email } });
    if (existing) return res.status(400).json({ error: 'This email is already registered. Please log in.' });

    const code = Math.floor(100000 + Math.random() * 900000).toString();
    pendingCodes.set(email, { code, expiresAt: Date.now() + 10 * 60 * 1000, token });

    const sent = await sendVerificationCode(email, code);
    if (!sent) {
      console.log('═══════════════════════════════════════════');
      console.log(`[DEV] REGISTRATION CODE for ${email}: ${code}`);
      console.log('═══════════════════════════════════════════');
    }

    res.json({ success: true, message: 'Verification code sent' });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: err.message });
  }
});

// Verify code + create account
router.post('/employee/register', async (req, res) => {
  try {
    const { email, code, name, phone, password, token } = req.body;

    const record = pendingCodes.get(email);
    if (!record) return res.status(401).json({ error: 'No verification code found' });
    if (record.code !== code) return res.status(401).json({ error: 'Incorrect code' });
    if (record.expiresAt < Date.now()) {
      pendingCodes.delete(email);
      return res.status(401).json({ error: 'Code expired' });
    }
    if (record.token !== token) return res.status(401).json({ error: 'Token mismatch' });

    const errors = validatePassword(password);
    if (errors.length) return res.status(400).json({ error: 'Password policy', details: errors });

    const invite = await prisma.employeeInvite.findUnique({ where: { token } });
    if (!invite || !invite.active || invite.expiresAt < new Date()) {
      return res.status(410).json({ error: 'Registration link expired' });
    }

    const existing = await prisma.user.findUnique({ where: { email } });
    if (existing) return res.status(400).json({ error: 'Email already registered' });

    const hashed = await bcrypt.hash(password, 10);
    const user = await prisma.user.create({
      data: { email, password: hashed, name, phone, role: 'EMPLOYEE', active: true },
    });

    await prisma.employeeInvite.update({
      where: { id: invite.id },
      data: { usedBy: user.id, active: false },
    });

    pendingCodes.delete(email);

    await logActivity({
      userId: user.id,
      action: 'EMPLOYEE_REGISTERED',
      interface: 'web',
      details: email,
      ip: req.ip,
    });

    res.json({ success: true, userId: user.id, email: user.email });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: err.message });
  }
});

module.exports = router;
